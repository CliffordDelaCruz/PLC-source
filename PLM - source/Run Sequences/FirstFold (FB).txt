FUNCTION_BLOCK FirstFold
VAR_INPUT
	Enable : BOOL;
END_VAR
VAR_OUTPUT
	Done : BOOL;
	Busy : BOOL;
	Fail : BOOL;
END_VAR
VAR
	lastEnable : BOOL := FALSE;
	nPhase : INT := 0;
END_VAR


// Rising edge detection
IF Enable AND NOT lastEnable THEN
	nPhase := 0;
	Done := FALSE;
	Busy := FALSE;
	Fail := FALSE;
END_IF

CASE nPhase OF
	
	0: 

	10: // Front Clamp up
	
 	20: // Retract front assembly by flange length
	
	30: // Front Clamp Down
	
	40: // Move to flange lenth plus x from back 
	
	50: // bend flange down 
	
	60: // bend flange down continued 
	
	
END_CASE

// Sets last enable value for rising edge detection
lastEnable := Enable;
