FUNCTION_BLOCK MiddleFold_TEST
VAR_INPUT
	ElementSize : LREAL;
	Enable : BOOL;
END_VAR
VAR_OUTPUT
END_VAR
VAR
	lastEnable : BOOL :=FALSE;
	TESTGO : BOOL := FALSE;
	TESTGO2 : BOOL := TRUE;
	fbCClampCtrl 	: CClampCtrl;
	fbFClampCtrl 	: FClampCtrl;
	fbBLClampCtrl 	: BLClampCtrl;
	fbBTClampCtrl 	: BTClampCtrl;
	foldTimeout 	: TON;
	fbGantryAbs 	: MC_MoveAbsolute;
	fbGantryHalt	: MC_Halt;
	fbBladeAbs		: MC_MoveAbsolute;
	fbBladeHalt 	: MC_Halt;
	fbLLCtrl		: LLControl;
	delayTimer		: TON;
	delayTimer2		: TON;
	fbFoldPathing   : FoldPathing_TEST;
	startPosTimeout : TON;
END_VAR
VAR
	nPhase : INT := 0;
	bCClampExe : BOOL := FALSE;
	bFClampExe : BOOL := FALSE;
	CClampDir  : ClampDirections;
	FClampDir  : ClampDirections;
	bBLClampExe : BOOL := FALSE;
	BLClampDir  : ClampDirections;
	bBTClampExe : BOOL := FALSE;
	BTClampDir  : ClampDirections;
	Busy : BOOL := FALSE;
	Fail : BOOL := FALSE;
	Done : BOOL := FALSE;
	LLTgtPos : LREAL;
	LLTgtVel : DINT;
	bLLMoveExe : BOOL := FALSE;
	bLastLLMoveExe : BOOL := FALSE;
	bDelay : BOOL := FALSE; 
	bGantryAbsExe : BOOL := FALSE;
	GantryTgtPos	: LREAL;
	GantryTgtVel	: LREAL;
	bBladeAbsExe	: BOOL := FALSE;
	BladeTgtPos	: LREAL;
	BladeTgtVel	: LREAL;
	n : INT := 0;
	startPosExe : BOOL;
	foldPathingExe 	: BOOL := FALSE;
	LastFoldPathingExe 	: BOOL := FALSE;
	MiddleFoldCounter1 : INT := 0;
	MiddleFoldCounter2 : INT := 0;
	bladeThickness : LREAL := 3.0;
	fbGantryVelocityTest : FB_GantryMoveVelocityTest;
    bRunVelocityTest     : BOOL := FALSE;
	bAbortVelocityTest : BOOL;
END_VAR


// Rising edge detection
IF Enable AND NOT lastEnable THEN
	nPhase := 0;
	Done := FALSE;
	Busy := FALSE;
	Fail := FALSE;
	
	bGantryAbsExe := FALSE;
    bBladeAbsExe := FALSE;
    bLLMoveExe := FALSE;
    bRunVelocityTest := FALSE;
	bAbortVelocityTest := FALSE;

    startPosTimeout(IN := FALSE);
    foldTimeout(IN := FALSE);
    delayTimer(IN := FALSE);
    delayTimer2(IN := FALSE);	
END_IF

CASE nPhase OF 
	0: // Start
		IF Enable THEN
			nPhase := 10;
			Busy := TRUE;
		END_IF
		
	10: // Move Axis to start
		IF NOT fbGantryAbs.Busy AND NOT fbLLCtrl.Busy AND NOT fbBladeAbs.Busy THEN
			bGantryAbsExe := TRUE;
			bBladeAbsExe := TRUE;
			bLLMoveExe := TRUE;
			// Target starting positions calculated based on element height
			LLTgtPos := 394;
			LLTgtVel := 50;
			GantryTgtPos := 20;
			GantryTgtVel := 40;
			BladeTgtPos := -5;
			BladeTgtVel	:= 40;
			startPosTimeout(IN := FALSE);
			nPhase := 11;
		ELSE
			startPosTimeout(IN := TRUE, PT := T#10S);
			IF startPosTimeout.Q THEN
				bGantryAbsExe := FALSE;
				bBladeAbsExe := FALSE;
				bLLMoveExe := FALSE;
				startPosTimeout(IN := FALSE);
				error.startPosTimeout := TRUE;
				Busy := FALSE;
				Fail := TRUE;
			END_IF
		END_IF
	
	11: // Move Axis to start continued
		MiddleFoldCounter1 := MiddleFoldCounter1 + 1;
		IF fbGantryAbs.Done AND fbLLCtrl.Done AND fbBladeAbs.Done THEN
			bGantryAbsExe := FALSE;
			bBladeAbsExe := FALSE;
			bLLMoveExe := FALSE;
			startPosTimeout(IN := FALSE);
			nPhase := 20;
		ELSIF fbGantryAbs.Error AND fbBladeAbs.Error THEN
			bGantryAbsExe := FALSE;
			bBladeAbsExe := FALSE;
			bLLMoveExe := FALSE;
			error.gantryMoveError := TRUE;
			error.bladeMoveError := TRUE;
			Busy := FALSE;
			Fail := TRUE;
		ELSIF fbGantryAbs.Error THEN
			bGantryAbsExe := FALSE;
			bBladeAbsExe := FALSE;
			bLLMoveExe := FALSE;
			error.gantryMoveError := TRUE;
			Busy := FALSE;
			Fail := TRUE;
		ELSIF fbBladeAbs.Error THEN
			bGantryAbsExe := FALSE;
			bBladeAbsExe := FALSE;
			bLLMoveExe := FALSE;
			error.bladeMoveError := TRUE;
			Busy := FALSE;
			Fail := TRUE;	 
		ELSE
			startPosTimeout(IN := TRUE, PT := T#20S);
			MiddleFoldCounter2 := MiddleFoldCounter2 + 1;
			IF startPosTimeout.Q THEN
				bGantryAbsExe := FALSE;
				bBladeAbsExe := FALSE;
				bLLMoveExe := FALSE;
				startPosTimeout(IN := FALSE);
				error.startPosTimeout := TRUE;
				Busy := FALSE;
				Fail := TRUE;
			END_IF
		END_IF
	
	20: // Move blade down to 0 as to be contacting material to start.
	IF NOT fbBladeAbs.Busy THEN
			bBladeAbsExe := TRUE;
			BladeTgtPos := 0;
			startPosTimeout(IN := FALSE);
			nPhase := 25;
		ELSE
			startPosTimeout(IN := TRUE, PT := T#10S);
			IF startPosTimeout.Q THEN
				bGantryAbsExe := FALSE;
				bBladeAbsExe := FALSE;
				bLLMoveExe := FALSE;
				startPosTimeout(IN := FALSE);
				error.startPosTimeout := TRUE;
				Busy := FALSE;
				Fail := TRUE;
			END_IF
		END_IF
		
	25: // Move blade down continued
		IF fbBladeAbs.Done THEN
			bBladeAbsExe := FALSE;
			bLLMoveExe := FALSE;
			startPosTimeout(IN := FALSE);
			nPhase := 60;
		ELSIF fbBladeAbs.Error THEN
			bGantryAbsExe := FALSE;
			bBladeAbsExe := FALSE;
			bLLMoveExe := FALSE;
			error.bladeMoveError := TRUE;
			Busy := FALSE;
			Fail := TRUE;	 
		ELSE
			startPosTimeout(IN := TRUE, PT := T#15S);
			IF startPosTimeout.Q THEN
				bGantryAbsExe := FALSE;
				bBladeAbsExe := FALSE;
				bLLMoveExe := FALSE;
				startPosTimeout(IN := FALSE);
				error.startPosTimeout := TRUE;
				Busy := FALSE;
				Fail := TRUE;
			END_IF
		END_IF
	
	60: // Fold the element
		delayTimer(IN := TRUE, PT := T#3S);
		IF delayTimer.Q
		AND NOT fbGantryVelocityTest.Busy
		AND NOT fbGantryAbs.Busy
		AND NOT bGantryAbsExe 
		THEN 
			delayTimer(IN := FALSE);
			bAbortVelocityTest := FALSE;
			bRunVelocityTest := TRUE;
			FoldTimeout(IN := FALSE);
			nPhase := 65;
		ELSE	
			foldTimeout(IN := TRUE, PT := T#60S);
			IF foldTimeout.Q THEN
				bRunVelocityTest := FALSE;
				foldTimeout(IN := FALSE);
				error.foldTimeout := TRUE;
				Busy := FALSE;
				Fail := TRUE;
			END_IF
		END_IF
		
	65: // Fold the element continued
		IF fbGantryVelocityTest.Done THEN
			bRunVelocityTest := FALSE;
			foldTimeout(IN := FALSE);
			nPhase := 170;
		ELSIF fbGantryVelocityTest.Error THEN
			bRunVelocityTest := FALSE;
			bAbortVelocityTest := TRUE;
			foldTimeout(IN := FALSE);
			error.pathingError := TRUE;
			Busy := FALSE;
			Fail := TRUE;
		ELSE 
			foldTimeout(IN := TRUE, PT := T#50S);
			IF foldTimeout.Q THEN
				bRunVelocityTest := FALSE;
				bAbortVelocityTest := TRUE;
				foldTimeout(IN := FALSE);
				error.foldTimeout := TRUE;
				Busy := FALSE;
				Fail := TRUE;
			END_IF
		END_IF
		
	170: // Reset to starting position
		foldTimeout(IN := FALSE);
		delaytimer2(IN := TRUE, PT := T#2S);
		IF delaytimer2.Q THEN
			nPhase := 0;
			delayTimer2(IN := FALSE);
			Done := TRUE;
		END_IF
	
	
END_CASE

// Gantry and blade movement fb calls
fbGantryAbs(Axis := GantryX1, Execute := bGantryAbsExe AND NOT bRunVelocityTest, Position := GantryTgtPos, Velocity := GantryTgtVel);
fbBladeAbs(Axis := BladeY1, Execute := bBladeAbsExe, Position := BladeTgtPos, Velocity := BladeTgtVel);

IF bLLMoveExe OR bLastLLMoveExe THEN 
	fbLLCtrl(fTargetPosition := LLTgtPos, nTargetSpeed := LLTgtVel, bMoveTrigger := bLLMoveExe);
END_IF

bLastLLMoveExe := bLLMoveExe;

fbGantryVelocityTest(
    Axis                := GantryX1,
    Execute             := bRunVelocityTest AND NOT bGantryAbsExe,
    SlowVelocity        := 5.0,
    FastVelocity        := 30.0,
    SlowZoneHalfWidth   := 25.0,
    Acceleration        := 17.0,
    Deceleration        := 25.0,
    Jerk                := 0.0,
    UpdateEveryNCycles  := 5,
	Abort				:= bAbortVelocityTest
);

// Sets last enable for rising edge detection
lastEnable := Enable;
