FUNCTION_BLOCK CutterStartup
VAR_INPUT
    bEnable     : BOOL;
END_VAR
VAR_OUTPUT
    Done        : BOOL;
    Busy        : BOOL;
    Fail        : BOOL;
END_VAR
VAR
	bLastEnable     : BOOL := FALSE;
	bExePower       : BOOL := FALSE;
	bExeSetPos      : BOOL := FALSE;
    bExeHome        : BOOL := FALSE;
    startupTimeout  : TON;
    nPhase          : INT := 0;
	homePosition	: LREAL := RetainedCtrlGVL.cutterHomePos + RetainedCtrlGVL.cutterHomePosOffset;
    fbPowerZ        : MC_Power;
    fbSetPosZ       : MC_SetPosition;
    fbHomeZ         : MC_Home;
	fbStopZ			: MC_Stop;
END_VAR


// This FB does the startup for the cutter axis

// Resets the fb
IF NOT bLastEnable AND bEnable THEN
	Busy := FALSE;
	Done := FALSE;
	Fail := FALSE;
	nPhase := 0;
END_IF

CASE nPhase OF
	0: // Wait for enable
		IF bEnable THEN
			nPhase := 10;
			Busy := TRUE;
		END_IF
		
	10: // Powers on axis, this also checks for something taking too long
		IF NOT fbPowerZ.Busy THEN
			bExePower := TRUE;
			startupTimeout(IN := FALSE);
			nPhase := 11;
		ELSE	
			startupTimeout(IN := TRUE, PT := T#5S);
			IF startupTimeout.Q THEN
				bExePower := FALSE;
				startupTimeout(IN := FALSE);
				error.cutterStartupTimeout := TRUE;
				Busy := FALSE;
				Fail := TRUE;
			END_IF
		END_IF

	11: // Checks for power status to allow moving to next step, also checks for errors and timeouts
		IF fbPowerZ.Status THEN
			startupTimeout(IN := FALSE);
			nPhase := 20;
		ELSIF fbPowerZ.Error THEN
		    bExePower := FALSE;
		 	error.cutterZ1Error := TRUE;
			Busy := FALSE;
			Fail := TRUE;
		ELSE	
			startupTimeout(IN := TRUE, PT := T#5S);
			IF startupTimeout.Q THEN
				bExePower := FALSE;
				startupTimeout(IN := FALSE);
				error.cutterStartupTimeout := TRUE;
				Busy := FALSE;
				Fail := TRUE;
			END_IF
		END_IF
		
	20: // Sets position to zero, Also checks for timeout.
		IF NOT fbSetPosZ.Busy AND NOT fbSetPosZ.Busy THEN
			bExeSetPos := TRUE;
			startupTimeout(IN := FALSE);
			nPhase := 21;
		ELSE	
			startupTimeout(IN := TRUE, PT := T#5S);
			IF startupTimeout.Q THEN
				bExeSetPos := FALSE;
				startupTimeout(IN := FALSE);
				error.cutterStartupTimeout := TRUE;
				Busy := FALSE;
				Fail := TRUE;
			END_IF
		END_IF

	21: // Checks for position set status to move to next step, also checks for errors and for timeouts
		IF fbSetPosZ.Done THEN
			bExeSetPos := FALSE;
			startupTimeout(IN := FALSE);
			nPhase := 30;
		ELSIF fbSetPosZ.Error THEN
			bExeSetPos := FALSE;
			error.cutterZ1Error := TRUE;
			Busy := FALSE;
			Fail := TRUE;
		ELSE	
			startupTimeout(IN := TRUE, PT := T#10S);
			IF startupTimeout.Q THEN
				bExeSetPos := FALSE;
				startupTimeout(IN := FALSE);
				error.cutterStartupTimeout := TRUE;
				Busy := FALSE;
				Fail := TRUE;
			END_IF
		END_IF
		
	30: // Home CutterZ1 axis, this also checks for something taking too long
		IF NOT fbHomeZ.Busy THEN
			bExeHome := TRUE;
			startupTimeout(IN := FALSE);
			nPhase := 31;
		ELSE	
			startupTimeout(IN := TRUE, PT := T#15S);
			IF startupTimeout.Q THEN
				bExeHome := FALSE;
				startupTimeout(IN := FALSE);
				error.cutterStartupTimeout := TRUE;
				Busy := FALSE;
				Fail := TRUE;
			END_IF
		END_IF

	31: // Checks for home status to allow moving to next step, also checks for errors and for timeouts
		IF fbHomeZ.Done THEN
			bExeHome := FALSE;
			startupTimeout(IN := FALSE);
			nPhase := 80;
		ELSIF fbHomeZ.Error THEN 
			bExeHome := FALSE;
			error.cutterHomeError := TRUE;	
			Busy := FALSE;
			Fail := TRUE;		
		ELSE	
			startupTimeout(IN := TRUE, PT := T#40S);
			IF startupTimeout.Q THEN
				bExeHome := FALSE;
				startupTimeout(IN := FALSE);
				error.cutterStartupTimeout := TRUE;
				Busy := FALSE;
				Fail := TRUE;
			END_IF
		END_IF
		
	80: // The cutter axis has successfully started up
		Busy := FALSE;
		Done := TRUE;
		Fail := FALSE;
		
END_CASE

// Power on fb call
fbPowerZ(Axis := CutterZ1, Enable := bExePower, Enable_Positive := bExePower, Enable_Negative := bExePower, Override := 100);

// Stops The motors if the startup fails
fbStopZ(Axis := CutterZ1, Execute := Fail);

// Set position fb call
fbSetPosZ(Axis := CutterZ1, Execute := bExeSetPos, Position := homePosition);

// Axis home fb call
fbHomeZ(Axis := CutterZ1, Execute := bExeHome, bCalibrationCam := DigInputs.Input3);

// Sets last enable trigger
bLastEnable := bEnable;
